import json
from pathlib import Path

def analyze_data(filepath: str) -> dict:
    # 读取并分析 JSON 数据文件
    data = json.loads(Path(filepath).read_text())
    result = {
        "total": len(data),
        "keys": list(data[0].keys()) if data else [],
    }
    return result
